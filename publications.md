<script>
  // Publications filters: "1st author only" toggle + venue checkboxes.
  // Everything is derived from the list itself: an entry is first-author when
  // it starts with the bold name, and its venue is the italic text, normalized
  // so repeat venues (ICRA 2026 / ICRA 2026 LBR, AAAI 2026, ...) collapse to one.
  (function () {
    var tools = document.getElementById('pub-tools');
    if (!tools) return;
    var main = tools.closest('.content') || document.body;

    function venueOf(li) {
      var em = li.querySelector('em');
      if (!em) return null;
      var t = em.textContent.replace(/\s+/g, ' ').trim();
      var workshop = /^workshop\b/i.test(t);
      if (t.indexOf(',') > -1) t = t.slice(t.lastIndexOf(',') + 1);  // "Workshop on ..., AAAI 2026"
      t = t.replace(/\b(19|20)\d{2}년?/g, ' ');                       // drop years
      var abbr = t.match(/\(([A-Za-z][A-Za-z&\-]{1,9})\)/);           // "(RA-L)", "(ICRA)"
      if (abbr) t = abbr[1];
      t = t.replace(/\([^)]*\)/g, ' ').replace(/\s+/g, ' ').trim();   // "(Late-Breaking Results)"
      return workshop ? t + ' Workshop' : t;
    }

    // Sections: each h2 owns the list items until the next h2.
    var sections = [];
    var heads = main.querySelectorAll('h2');
    for (var i = 0; i < heads.length; i++) {
      var sec = { head: heads[i], blocks: [], items: [] };
      for (var n = heads[i].nextElementSibling; n && n.tagName !== 'H2'; n = n.nextElementSibling) {
        sec.blocks.push(n);
        var lis = n.matches('ul') ? n.children : n.querySelectorAll('li');
        for (var k = 0; k < lis.length; k++) {
          var li = lis[k];
          var first = li.firstChild;
          while (first && first.nodeType === 3 && !first.textContent.trim()) first = first.nextSibling;
          sec.items.push({
            el: li,
            first: !!(first && first.nodeType === 1 && first.tagName === 'STRONG'),
            venue: venueOf(li)
          });
        }
      }
      sections.push(sec);
    }

    var all = [];
    sections.forEach(function (s) { all = all.concat(s.items); });
    if (!all.length) return;

    // Unique venues, most frequent first.
    var counts = {};
    all.forEach(function (it) { if (it.venue) counts[it.venue] = (counts[it.venue] || 0) + 1; });
    var venues = Object.keys(counts).sort(function (a, b) {
      return counts[b] - counts[a] || a.localeCompare(b);
    });

    var list = document.getElementById('pub-venue-list');
    venues.forEach(function (v) {
      var lab = document.createElement('label');
      lab.className = 'pub-venue';
      var cb = document.createElement('input');
      cb.type = 'checkbox';
      cb.checked = true;
      cb.value = v;
      var name = document.createElement('span');
      name.textContent = v;
      var c = document.createElement('span');
      c.className = 'pub-venue-n';
      c.textContent = counts[v];
      lab.appendChild(cb); lab.appendChild(name); lab.appendChild(c);
      list.appendChild(lab);
    });
    var boxes = list.querySelectorAll('input');

    var firstBtn = document.getElementById('pub-first');
    var venueBtn = document.getElementById('pub-venue-btn');
    var panel = document.getElementById('pub-venue-panel');
    var badge = document.getElementById('pub-venue-count');
    var status = document.getElementById('pub-status');
    var firstOnly = false;

    function apply() {
      var on = {}, nOn = 0;
      for (var i = 0; i < boxes.length; i++) if (boxes[i].checked) { on[boxes[i].value] = 1; nOn++; }
      var venueFiltered = nOn < boxes.length;
      var filtering = firstOnly || venueFiltered;
      var shown = 0;
      sections.forEach(function (s) {
        var vis = 0;
        s.items.forEach(function (it) {
          var ok = (!firstOnly || it.first) && (!venueFiltered || (it.venue && on[it.venue]));
          it.el.hidden = !ok;
          if (ok) vis++;
        });
        shown += vis;
        var hide = filtering && vis === 0;
        s.head.hidden = hide;
        s.blocks.forEach(function (b) { b.hidden = hide; });
      });
      firstBtn.setAttribute('aria-pressed', firstOnly ? 'true' : 'false');
      badge.textContent = venueFiltered ? nOn + '/' + boxes.length : '';
      venueBtn.classList.toggle('is-filtered', venueFiltered);
      status.innerHTML = filtering
        ? '<span class="lang-en">Showing ' + shown + ' of ' + all.length + '</span>' +
          '<span class="lang-ko">' + all.length + '편 중 ' + shown + '편 표시</span>'
        : '';
    }

    firstBtn.addEventListener('click', function () { firstOnly = !firstOnly; apply(); });
    venueBtn.addEventListener('click', function () {
      var open = venueBtn.getAttribute('aria-expanded') !== 'true';
      venueBtn.setAttribute('aria-expanded', open ? 'true' : 'false');
      panel.classList.toggle('open', open);
    });
    list.addEventListener('change', apply);
    var bulk = tools.querySelectorAll('[data-venue-all]');
    for (var b = 0; b < bulk.length; b++) {
      bulk[b].addEventListener('click', function () {
        var v = this.getAttribute('data-venue-all') === '1';
        for (var i = 0; i < boxes.length; i++) boxes[i].checked = v;
        apply();
      });
    }

    tools.hidden = false;
    apply();
  })();
</script>
